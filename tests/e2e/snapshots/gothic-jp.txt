 * 下面只定义需要覆盖它的类别：用 unicode-range 接管字符，用 size-adjust 调整大小。

/* 斜体：每组规则另有一份 font-style: italic。有斜体的类别用斜体字体（或倾斜轴）；
 * 没有斜体的类别（汉字、假名等）用正体文件、也声明成 italic，浏览器就不会把它们模拟倾斜。 */
/* ascent-override / descent-override / line-gap-override：西文等规则使用汉字字体的行高度量（已除以 size-adjust），
 * 行高和基线位置都和汉字字体一致。Chrome 87+、Firefox 89+ 支持；Safari 会忽略，按各字体自己的度量显示。 */

/* 复合字体「My Composite」· 基线偏移
 * @font-face 没有基线描述符，这段脚本把需要偏移的字符包进 <span>，
 * 用相对定位移动，不影响行高。放在页面末尾，或在内容渲染后调用。 */
(() => {
  const SELECTOR = '.composite';
  const SHIFT = [ // [起, 止, 偏移 em]，正值向上
    [0x0020, 0x007E, -0.02], [0x00A0, 0x00B6, -0.02], [0x00B8, 0x02C6, -0.02],
    [0x02C8, 0x02C9, -0.02], [0x02CC, 0x02D8, -0.02], [0x02DA, 0x04FF, -0.02],
    [0x1E00, 0x1EFF, -0.02], [0x2000, 0x2013, -0.02], [0x2016, 0x2017, -0.02],
    [0x201A, 0x201B, -0.02], [0x201E, 0x2024, -0.02], [0x2027, 0x203A, -0.02],
    [0x203C, 0x206F, -0.02], [0x20A0, 0x20CF, -0.02], [0x2122, 0x2122, -0.02],
    [0xFB00, 0xFB06, -0.02]
  ];
  const shiftOf = cp => {
    let lo = 0, hi = SHIFT.length - 1;
    while (lo <= hi) {
      const m = (lo + hi) >> 1, r = SHIFT[m];
      if (cp < r[0]) hi = m - 1; else if (cp > r[1]) lo = m + 1; else return r[2];
    }
    return 0;
  };
  const skip = el => !el || el.closest('[data-cf-shift], script, style, textarea, code, pre');
  function apply(root) {
    const walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT);
    const nodes = [];
    while (walker.nextNode()) if (!skip(walker.currentNode.parentElement)) nodes.push(walker.currentNode);
    for (const node of nodes) {
      const runs = [];
      let cur = null;
      for (const ch of node.data) {
        const s = shiftOf(ch.codePointAt(0));
        if (cur && cur.s === s) cur.t += ch; else runs.push(cur = { s, t: ch });
      }
      if (!runs.some(r => r.s)) continue;
      const frag = document.createDocumentFragment();
      for (const r of runs) {
        if (!r.s) { frag.append(r.t); continue; }
        const span = document.createElement('span');
        span.dataset.cfShift = '';
        span.style.cssText = `position:relative;inset-block-start:${-r.s}em`;
        span.textContent = r.t;
        frag.append(span);
      }
      node.replaceWith(frag);
    }
  }
  document.querySelectorAll(SELECTOR).forEach(apply);
})();
